test("delayed interaction updates a counteroffer only when due",()=>{m.negotiations=[{id:"n1",name:"Deal",when:2,initialOffer:5,targetOffer:8,rounds:1,concessionRate:0}];m.decisionInteractions=[{id:"i1",name:"Counter",actorId:"a1",negotiationId:"n1",actionType:"counteroffer",when:1,delaySteps:2,offerValue:7,objectionStatus:"open"}];const r=STRATOSSimulation.simulate(m,{steps:3,seed:"counter"});return r.decisionInteractions[0].step===3&&r.negotiations[0].rounds.length===1&&r.negotiations[0].finalOffer>=7;}); test("withdrawal blocks a scheduled negotiation and return reopens it",()=>{m.negotiations=[{id:"n1",name:"Deal",when:2,initialOffer:5,targetOffer:5,rounds:1,concessionRate:0}];m.decisionInteractions=[{id:"i1",name:"Withdraw",negotiationId:"n1",actionType:"withdraw",when:1,delaySteps:0},{id:"i2",name:"Return",negotiationId:"n1",actionType:"return",when:3,delaySteps:0}];const r=STRATOSSimulation.simulate(m,{steps:3,seed:"return"});return r.negotiations.length===1&&r.negotiations[0].step===3&&r.decisionInteractions.length===2;}); test("objection remains open unless explicitly marked addressed",()=>{m.decisionInteractions=[{id:"i1",name:"Price concern",actionType:"raise_objection",when:1,objection:"Price",objectionStatus:"open"},{id:"i2",name:"Evidence supplied",actionType:"request_evidence",when:2,objection:"Price",objectionStatus:"addressed",responseOrEvidence:"Comparable quote"}];const r=STRATOSSimulation.simulate(m,{steps:2,seed:"objection"});return r.decisionInteractions[0].objectionStatus==="open"&&r.decisionInteractions[1].objectionStatus==="addressed";}); test("delayed interaction effects are applied at their action step",()=>{m.variables[0].level=5;m.decisionInteractions=[{id:"i1",name:"Return",actionType:"return",when:1,delaySteps:2,effects:"v1:+2"}];const r=STRATOSSimulation.simulate(m,{steps:3,seed:"effect"});return r.timeline[0].levels.v1===5&&r.timeline[2].levels.v1>5;}); test("decision interaction beyond horizon is not falsely reported as completed",()=>{m.decisionInteractions=[{id:"i1",name:"Later reply",actionType:"return",when:4,delaySteps:3}];const r=STRATOSSimulation.simulate(m,{steps:4,seed:"horizon"});return r.decisionInteractions.length===0&&r.warnings.some(x=>x.includes("scheduled interaction"));});
